action(PRED)
IF		?ty(e>t)
THEN	make(\/1)
        go(\/1)
        put(ty(e>(e>t)))
        ttrput(R1^R2^(R1 ++ (R2 ++ [p1==state_PRED(R1.head,R2.head):t])))
        put(!)
        go(/\1)
        make(\/0)
        go(\/0)
        put(?ty(e))
